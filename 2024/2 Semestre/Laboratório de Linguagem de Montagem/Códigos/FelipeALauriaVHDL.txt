library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity example2 is
    Port (
        x1 : in std_logic;
        x2 : in std_logic;
        x3 : in std_logic;
        x4 : in std_logic;
        f : out std_logic;
        g : out std_logic;
        h : out std_logic
    );
end example2;

architecture Behavioral of example2 is
begin
    g <= (x1 and x3) or (x2 and x4);
    h <= (x1 and not x3) or (not x2 and x4);
    f <= g or h;
end Behavioral;
